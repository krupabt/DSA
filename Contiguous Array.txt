class Solution:
    def findMaxLength(self, nums: list[int]) -> int:
        h = {}
        balance = 0
        ans = 0

        h[0] = -1

        for i in range(len(nums)):
            if nums[i] == 1:
                balance = balance + 1
            else:
                balance = balance - 1

            if balance in h:
                length = i - h[balance]

                if length > ans:
                    ans = length
            else:
                h[balance] = i

        return ans
