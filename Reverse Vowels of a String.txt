class Solution:
    def reverseVowels(self, s: str) -> str:
        s = list(s)

        i = 0
        j = len(s) - 1

        while i < j:

            while i < j:
                if s[i] in "AEIOUaeiou":
                    break
                i += 1

            while i < j:
                if s[j] in "AEIOUaeiou":
                    break
                j -= 1

            s[i], s[j] = s[j], s[i]

            i += 1
            j -= 1

        return "".join(s)
