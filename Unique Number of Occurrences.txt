class Solution:
    def uniqueOccurrences(self, arr: list[int]) -> bool:
        h = {}

        
        for num in arr:
            if num in h:
                h[num] = h[num] + 1
            else:
                h[num] = 1
 
        
        seen = set()

        for num in h:
            count = h[num]

            if count in seen:
                return False

            seen.add(count)

        return True
