
class Solution:
    def wordPattern(self, pattern: str, s: str) -> bool:
        h1 = {}
        h2 = {}

        words = s.split()

        if len(pattern) != len(words):
            return False

        for i in range(len(pattern)):
            if pattern[i] in h1:
                if h1[pattern[i]] != words[i]:
                    return False
            else:
                if words[i] in h2:
                    return False

                h1[pattern[i]] = words[i]
                h2[words[i]] = pattern[i]

        return True
