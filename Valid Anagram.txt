class Solution:
    def isAnagram(self, s: str, t: str) -> bool:
        h1 = {}
        h2 = {}

        for i in s:
            if i in h1:
                count = h1[i]
                count = count + 1
                h1[i] = count
            else:
                h1[i] = 1

        for i in t:
            if i in h2:
                count = h2[i]
                count = count + 1
                h2[i] = count
            else:
                h2[i] = 1

        return h1 == h2
